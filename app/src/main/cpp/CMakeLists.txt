cmake_minimum_required(VERSION 3.22.1)
project("ai_pipeline")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_definitions(-DMNN_OPENCL=ON)
add_definitions(-DMNN_LOW_MEMORY=ON)
add_definitions(-DNCNN_VULKAN=ON)
add_library(MNN SHARED IMPORTED)
set_target_properties(MNN PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/libs/${ANDROID_ABI}/libMNN.so)
add_library(ncnn STATIC IMPORTED)
set_target_properties(ncnn PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/libs/${ANDROID_ABI}/libncnn.a)
find_library(log-lib log)
find_library(jnigraphics-lib jnigraphics)
find_library(android-lib android)
find_library(opencl-lib OpenCL)
if(NOT opencl-lib)
set(opencl-lib OpenCL)
endif()
add_library(native_pipeline SHARED pipeline_jni.cpp diffusion_mnn.cpp upscale_ncnn.cpp image_processing.cpp)
target_include_directories(native_pipeline PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(native_pipeline MNN ncnn vulkan ${opencl-lib} ${log-lib} ${jnigraphics-lib} ${android-lib})
